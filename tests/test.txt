Option Compare Database
Option Explicit

Public Function RefreshDeviceDataTables() As Boolean
    On Error GoTo ErrHandler

    Dim wrk As DAO.Workspace
    Dim db As DAO.Database
    Dim inTransaction As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    Set wrk = DBEngine.Workspaces(0)
    Set db = CurrentDb()

    wrk.BeginTrans
    inTransaction = True

    db.Execute "DELETE * FROM DeviceData;", dbFailOnError
    CopyBranchDeviceData db

    db.Execute "DELETE * FROM TFMR_DeviceData;", dbFailOnError
    CopyTransformerDeviceData db

    wrk.CommitTrans
    inTransaction = False

    RefreshDeviceDataTables = True

CleanExit:
    Set db = Nothing
    Set wrk = Nothing
    If errNumber <> 0 Then
        On Error GoTo 0
        Err.Raise errNumber, "DeviceDataUpdater.RefreshDeviceDataTables", errDescription
    End If
    Exit Function

ErrHandler:
    errNumber = Err.Number
    errDescription = Err.Description
    If inTransaction Then
        wrk.Rollback
    End If
    RefreshDeviceDataTables = False
    Resume CleanExit
End Function

Private Sub CopyBranchDeviceData(ByVal db As DAO.Database)
    On Error GoTo ErrHandler

    Dim sourceRows As DAO.Recordset
    Dim targetRows As DAO.Recordset
    Dim branchId As Variant
    Dim targetHasRows As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    Set sourceRows = db.OpenRecordset("DeviceData_All", dbOpenSnapshot)
    Set targetRows = db.OpenRecordset("DeviceData", dbOpenDynaset, dbSeeChanges)
    targetHasRows = Not (targetRows.BOF And targetRows.EOF)

    Do While Not sourceRows.EOF
        branchId = sourceRows.Fields("branch_id").Value
        If targetHasRows Then
            targetRows.FindFirst "branch_id = " & branchId
        End If

        If Not targetHasRows Or targetRows.NoMatch Then
            targetRows.AddNew
            targetRows.Fields(0).Value = sourceRows.Fields("branch_id").Value
            targetRows.Fields(1).Value = sourceRows.Fields("nostat").Value
            targetRows.Fields(2).Value = sourceRows.Fields("fromnum").Value
            targetRows.Fields(3).Value = sourceRows.Fields("tonum").Value
            targetRows.Fields(4).Value = sourceRows.Fields("fromtype").Value
            targetRows.Fields(5).Value = sourceRows.Fields("totype").Value
            targetRows.Fields(6).Value = sourceRows.Fields("bus1").Value
            targetRows.Fields(7).Value = sourceRows.Fields("bus2").Value
            targetRows.Fields(8).Value = sourceRows.Fields("ckt").Value
            targetRows.Fields(9).Value = sourceRows.Fields("bus1Name").Value
            targetRows.Fields(10).Value = sourceRows.Fields("bus2Name").Value
            targetRows.Fields(11).Value = sourceRows.Fields("line_name").Value
            targetRows.Fields(12).Value = sourceRows.Fields("start_date").Value
            targetRows.Fields(13).Value = sourceRows.Fields("end_date").Value
            targetRows.Update
            targetHasRows = True
        Else
            targetRows.Edit
            If Nz(sourceRows.Fields("fromnum").Value, "") <> "" Then
                targetRows.Fields("fromnum").Value = sourceRows.Fields("fromnum").Value
            End If
            If Nz(sourceRows.Fields("tonum").Value, "") <> "" Then
                targetRows.Fields("tonum").Value = sourceRows.Fields("tonum").Value
            End If
            targetRows.Update
        End If

        sourceRows.MoveNext
    Loop

CleanExit:
    On Error Resume Next
    If Not targetRows Is Nothing Then
        targetRows.Close
    End If
    If Not sourceRows Is Nothing Then
        sourceRows.Close
    End If
    Set targetRows = Nothing
    Set sourceRows = Nothing
    If errNumber <> 0 Then
        On Error GoTo 0
        Err.Raise errNumber, "DeviceDataUpdater.CopyBranchDeviceData", errDescription
    End If
    Exit Sub

ErrHandler:
    errNumber = Err.Number
    errDescription = Err.Description
    Resume CleanExit
End Sub
